    OUT

; в AX передаем текущее значение n, 
; в конце работы результат будет в DX
:factorial 
        PUSH AX 
        PUSH 1
        JBE :base_factorial

        PUSH AX
        DEC AX

        CALL :factorial

        PUSH DX 
        MUL
        POP DX 

        RET

    :base_factorial
        PUSH 1
        POP DX
        RET

    JMP :exit
